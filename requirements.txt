# AURELIX — the deployable dependency set (API + agent pipeline).
#
# `agent_core/requirements.txt` and `platform_backend/requirements.txt` describe the two
# halves separately and are kept for local, single-component work. A host installs one file,
# so this is that file, and it is the one Render builds from.
#
# Versions are the ones the 249-test suite actually passes against. Binary packages
# (numpy, Pillow, OpenCV) are floored rather than pinned so pip resolves a wheel for
# whatever interpreter the host runs — pinning them exactly is how a deploy ends up
# compiling NumPy from source on a 0.1-CPU free-tier container.

# ── API ─────────────────────────────────────────────────────────────────────
fastapi==0.137.2
uvicorn[standard]==0.49.0
python-multipart==0.0.32
sqlalchemy==2.0.51
pydantic==2.13.4
pydantic-settings==2.14.2
python-dotenv==1.2.2

# ── Agent pipeline ──────────────────────────────────────────────────────────
google-genai==2.9.0
PyYAML==6.0.3
rank-bm25==0.2.2
Pillow>=11.0
numpy>=2.0
# Headless: the Phase 4.1 quality gate needs cv2.Laplacian, not a GUI toolkit. The non-
# headless build pulls in X11 shared libraries that are not present in a slim container.
opencv-python-headless>=4.10

# ── Text gateway (Phase 7) ──────────────────────────────────────────────────
# The official OpenAI SDK, used for Groq's OpenAI-compatible endpoint (free tier) and for
# OpenAI itself when a key is set. Imported lazily, only on the first call with a key:
# measured +17.6 MB peak RSS when loaded (113.2 -> 130.8 MB for the whole app). Pinned to
# the version whose source the adapter was written against.
openai==2.43.0

# ── Policy Copilot retrieval (Phase 7) ──────────────────────────────────────
# A local ONNX sentence-embedding model (BAAI/bge-small-en-v1.5, 67 MB, MIT), run on CPU.
# No API, no quota. Wheels verified for linux x86_64 / CPython 3.13 (Render) and 3.14.
# The model is downloaded once by `python -m agent_core.tools.build_index` during the build
# and embeds the policy there; the server only embeds questions. Measured serving footprint:
# 287 MB for the whole API with the model loaded (evaluation/rag/report.md).
fastembed==0.8.1
onnxruntime==1.30.0

# ── Authentication (Phase 7) ────────────────────────────────────────────────
# JWT access tokens (HS256) and Argon2id password hashing at the OWASP minimum
# (19 MiB, t=2, p=1). Pure-Python / abi3 wheels for linux x86_64 CPython 3.13.
pyjwt==2.15.1
argon2-cffi==25.1.0
